import { it } from "vitest";
import { JOURNEY_IMAGE_PALETTES } from "@/lib/particles/journey-palettes.generated";
import { particlePaletteFromImage } from "@/lib/journeys/particle-lead";
import { readFileSync } from "node:fs";
const names: Record<string, string> = {};
for (const m of readFileSync("src/lib/journeys/particle-profiles.generated.ts", "utf8").matchAll(/"([^"]+)": \{"name":"([^"]+)"/g)) names[m[1]] = m[2];
const hsv = (r: number, g: number, b: number) => { const mx = Math.max(r, g, b), mn = Math.min(r, g, b), d = mx - mn; let h = 0; if (d > 1e-6) h = mx === r ? ((g - b) / d) % 6 : mx === g ? (b - r) / d + 2 : (r - g) / d + 4; return [((h / 6) + 1) % 1, mx ? d / mx : 0, mx]; };
const lin2s = (c: number) => (c <= 0.0031308 ? 12.92 * c : 1.055 * Math.pow(c, 1 / 2.4) - 0.055);
const warm = (h: number) => h < 0.11 || h > 0.88; // red/orange/pink/magenta-pink
it("audit", () => {
let tot = 0, tw = 0;
const rows: [number, number, string][] = [];
for (const [id, ip] of Object.entries(JOURNEY_IMAGE_PALETTES)) {
  let n = 0, w = 0, imgW = 0, imgT = 0;
  for (const ph of ip.phases.length ? ip.phases : [ip]) {
    ph.colors.forEach((c, i) => { const x = parseInt(c.slice(1), 16); const [h, s, v] = hsv(((x >> 16) & 255) / 255, ((x >> 8) & 255) / 255, (x & 255) / 255); if (s > 0.15 && v > 0.15) { imgT += ph.weights[i]; if (warm(h)) imgW += ph.weights[i]; } });
    for (let v = 0; v < 4; v++) {
      const p = particlePaletteFromImage(ph, v); if (!p) continue;
      for (const c of [p.low, p.mid, p.high]) { const [h] = hsv(lin2s(c[0]), lin2s(c[1]), lin2s(c[2])); n++; if (warm(h)) w++; }
    }
  }
  tot += n; tw += w;
  rows.push([w / Math.max(1, n), imgT ? imgW / imgT : 0, names[id] ?? id]);
}
rows.sort((a, b) => b[0] - a[0] - (b[1] - a[1]));
console.log("particles-warm  image-warm  journey");
for (const r of rows) console.log(`${(r[0] * 100).toFixed(0).padStart(5)}%  ${(r[1] * 100).toFixed(0).padStart(5)}%   ${r[2]}`);
console.log("overall particle stops warm:", (tw / tot * 100).toFixed(0) + "%");

});
